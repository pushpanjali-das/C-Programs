/* Program to set the string with given charcater.
   Also to set the string with the given charcater upto the given "n"  number of charcaters 
*/
#include<stdio.h>
#include<string.h>

int main()
{
    char x[40],ch;
    int n;
    printf("Enter the string\n");
    scanf("%[^\n]",&x);
    printf("Enter the character which will replace the string characters\n");
    scanf(" %c",&ch);
    printf("new string %s",strset(x,ch);
    printf("Enter the number of characters to be set\n");
    scanf("%d",&n);
    printf("new string %s",strset(x,ch,n);
    printf("program has ended\n");
    return 0;
}
