/* Program to convert lower case characters in the string into upper case */

#include<stdio.h>
int main()
{
    char x[40];
    int i=0;
    printf("Enter the string\n");
    scanf("%[^\n]",&x);                 
    while(x[i]!='\0') 
    {
        if(x[i]>=97 && x[i]<=122)  
        {
            x[i]=x[i]-32;           
        }
        i++;
    }
    printf("String in upper case is %s",x);
    return 0;
}
