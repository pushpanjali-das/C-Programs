#include<stdio.h>
int main()
{
    int n1,n2,temp1,temp2,r,newNum=0,pow=1,count1=0,count2=0;
    printf("Enter the 2 ranges\n");
    scanf("%d%d",&n1,&n2);
    while(n1<=n2)
    {
        temp1=n1;
        temp2=n1;
        count1=0;
        newNum=0;
        while(temp1>0)
        {
            count1++;
            temp1=temp1/10;
        }
        while(temp2>0)
        {
            count2=count1;
            pow=1;
            r=temp2%10;
            while(count2>0)
            {
                pow=pow*r;
                count2--;
            }
            temp2=temp2/10;
            newNum=newNum+pow;
        }
        if(newNum==n1)
        {
            printf("%d\n",n1);
        }
        n1++;
    }
    printf("program ended successfully\n");
    return 0;
}
