/* Program to count the number of vowels, consonants, digits and spaces present in the given string */

#include<stdio.h>
int main()
{
    char x[40];
    int vowels=0,alphabet=0,digit=0,spaces=0,i=0;
    printf("Enter the string\n");
    scanf("%[^\n]",&x);
    while(x[i]!='\0')
    {
        if(x[i]=='a'||x[i]=='e'||x[i]=='i'||x[i]=='o'||x[i]=='u'||x[i]=='A'||x[i]=='E'||x[i]=='I'||x[i]=='O'||x[i]=='U')
        {
            vowels++;           
        }
        if((x[i]>='a' && x[i]<='z') || (x[i]>='A' && x[i]<='Z'))
        {
            alphabet++;         
        }
        else if(x[i]==' ')
        {
            spaces++;           
        }
        else if(x[i]>='0' && x[i]<='9')
        {
            digit++;            
        }
        i++;
    }
    printf("Number of vowels is %d\n",vowels);
    printf("Number of consonants is %d\n",alphabet-vowels);
    printf("Number of digits is %d\n",digit);
    printf("Number of spaces is %d\n",spaces);
    return 0;
}
