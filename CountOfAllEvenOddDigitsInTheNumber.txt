/* Count of all, even and odd digits present in a given number */
#include<stdio.h>
int main()
{
    int n,r,count=0,counto=0,counte=0,temp;
    printf("Enter the number\n");
    scanf("%d",&n);
    temp=n;
    while(n>0)
    {
        r=n%10;
        count++;
        if(r%2!=0)
        {
            counto++;
        }
        else
        {
            counte++;
        }
        n=n/10;
    }
    printf("Count of all the digits in the given number %d is %d\nCount of all the even digits is %d\nCount of all the odd digits is %d",temp,count,counte,counto);
}
