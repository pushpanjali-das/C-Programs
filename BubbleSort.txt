/* Bubble Sort */
#include<stdio.h>
int main()
{
    int arr[]={34,5,612,24,889,8},size=0;
    size=sizeof(arr)/sizeof(int);
    for(int i=0;i<size-1;i++)
    {
        for(int j=0;j<size-i-1;j++)
        {
            if(arr[j]>arr[j+1])
            {
                    int temp =0;
                temp =arr[j];
                arr[j]=arr[j+1];
                arr[j+1]=temp;
            }
        }
    }
    printf("array after sorting in ascending order is\n");
    for(int i=0;i<size;i++)
    {
        printf("%d ",arr[i]);
    }
}
