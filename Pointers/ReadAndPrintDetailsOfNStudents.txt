/* Program to read "n" number of students roll number, name and marks in 3 subjects. Find the student total, average, result and the grade.
            Average                         Grade
          ~~~~~~~~~~                      ~~~~~~~~~~    
          >=60                              First grade
          >=50 and <60                      Second grade
          >=35 and <50                      Third garde
          <35                               No grade
*/
#include<stdio.h>
#include<string.h>
struct student
{
    int sno,tot,m1,m2,m3;
    char sname[40],result[10],grade[20];
    float average;
};
int main()
{
    struct student s[10],*p;
    int n;
    p=&s[0];
    printf("Enter the number of students\n");
    scanf("%d",&n);
    for(;p<&s[n];p++)
    {
        printf("Enter the student roll number, name and marks in 3 subjects\n");
        scanf("%d %[^\n]%d%d%d",&p->sno,p->sname,&p->m1,&p->m2,&p->m3);
        p->tot=p->m1+p->m2+p->m3;
        p->average=(float)p->tot/3;
        if(p->m1>=35 && p->m2>=35 && p->m3>=35)
        {
            strcpy(p->result,"Passed");
            if(p->average>=60)
            {
                strcpy(p->grade,"First grade");
            }
            else if(p->average>=50 && p->average<60)
            {
                strcpy(p->grade,"Second grade");
            }
            else
            {
                strcpy(p->grade,"Third grade");
            }
        }
        else
        {
            strcpy(p->result,"Failed");
            strcpy(p->grade,"No grade");
        }
    }
    printf("Students information\n");
    for(p=&s[0];p<&s[n];p++)
    {
        printf("%d  %s  %d   %.2f  %s  %s\n",p->sno,p->sname,p->tot,p->average,p->result,p->grade);
    }
    return 1;
}
