/* Program to print Lucky Digit of the given number 
    
    EX: 1947 = 1+9+4+7=21=2+1=3
             =3
*/
#include<stdio.h>
int main()
{
    int n,r,sum,temp;
    printf("Enter the number\n");
    scanf("%d",&n);
    temp=n;
    while(n>9)
    {
        sum=0;
        while(n>0)
        {
            r=n%10;
            sum=sum+r;
            n=n/10;
        }
        n=sum;
    }
    printf("Lucky digit for %d is %d",temp,n);
    return 0;
}
