/* Program to check whether the string contains given sub string or not

If sub string is present, sub string along with the rest of the string will be printed.

Else
    null will be returned.
    
*/

#include<stdio.h>
#include<string.h>

int main()
{
    char x[40],y[40];
    printf("Enter the string\n");
    scanf("%[^\n]",&x);
    printf("Enter the sub string\n");
    scanf(" %[^\n]",&y);
    printf("New string is %s",strstr(x,y));
    return 0;
}
