/* malloc function */

#include<stdio.h>
#include<stdlib.h>

int main()
{
    int n;
    printf("enter number of integers need to be stored\n");
    scanf("%d",&n);
    int *arr=(int *)malloc(n*sizeof(int));
    if(arr==NULL)
    {
        printf("Memory allocation did not happen\n");
    }
    else
    {
        printf("Memory has been allocated succesfully\n");
        printf("Enter array elements\n");
        for(int i=0;i<n;i++)
        {
            scanf("%d",&*(arr+i));
        }
        printf("Array elements are\n");
        for(int i=0;i<n;i++)
        {
            printf("%d ",*(arr+i));
        }
    }
}
