/* Program to set the string with the given character */

#include<stdio.h>
int main()
{
    char x[40],ch;
    int i;
    printf("Enter the string\n");
    scanf("%[^\n]",&x);
    printf("Enter the charcater\n");
    scanf(" %c",&ch);
    for(i=0;x[i]!='\0';i++)
    {
        x[i]=ch;
    }
    printf("New string is %s",x);
    return 0;
}
