/* Perfect Number 
    Ex: 6,28 etc
*/
#include<stdio.h>
int main()
{
    int n,i=1,sum=0;
    printf("Enter the number\n");
    scanf("%d",&n);
    while(i<n)
    {
        if(n%i==0)
        {
            printf("%d\n",i);
            sum=sum+i;
        }
        i++;
    }
    (sum==n)?printf("%d is a perfect number",n):printf("%d is not a perfect number",n);
}
