/* Changing the case of the String */

#include<stdio.h>
#include<string.h>
int main()
{
    char s[20];
    int length=0;
    printf("enter the string\n");
    scanf("%s",&s);
    for(int i=0;s[i]!='\0';i++)
    {
        
        if(((s[i]>=97) && (s[i]<=122)))
        {
            s[i]=s[i]-32;
        }
        else if(s[i]>=65 && s[i]<=90)
        {
            s[i]=s[i]+32;
        }
        length++;
    }
    printf("Updated string is %s and it's length is %d",s,length);
}
