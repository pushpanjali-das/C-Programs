/*  1 2 3 4 5
      1 2 3 4
        1 2 3
          1 2
            1
*/

#include<stdio.h>
int main()
{
    int n,i,j,sp;
    printf("Enter the value of n\n");
    scanf("%d",&n);
    for(i=n;i>=1;i--)
    {
        for(sp=i;n-sp>0;sp++)
        {
            printf("   ");
        }
        for(j=1;j<=i;j++)
        {
            printf("%-3d",j);
        }
        printf("\n");
    }
    return 0;
}
