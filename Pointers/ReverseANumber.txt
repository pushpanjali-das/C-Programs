/* Program to reverse a number */
#include<stdio.h>
int main()
{
    int n,*p,r,rev=0;
    p=&n;
    printf("Enter the number\n");
    scanf("%d",p);
    while((*p)>0)
    {
        r=(*p)%10;
        rev=(rev*10)+r;
        (*p)=(*p)/10;
    }
    printf("Reverse number is %d",rev);
    return 0;
}
