/* Program to print the Transpose of the given m*m (or) n*n matrix.

    a(2*3) = 1   2   3
             4   5   6
             
    b(3*2) = 1  4
             2  5
             3  6
*/

#include<stdio.h>
int main()
{
    int a[5][5],b[5][5],m,n,i,j;
    printf("Enter number of rows and columns\n");
    scanf("%d%d",&m,&n);
    for(i=0;i<m;i++)
    {
        for(j=0;j<n;j++)
        {
            printf("Enter value of a[%d][%d]\n");
            scanf("%d",&a[i][j]);
            b[j][i]=a[i][j];
        }
    }
    printf("Given matrix\n");
    for(i=0;i<m;i++)
    {
        for(j=0;j<n;j++)
        {
            printf("%d ",a[i][j]);
        }
        printf("\n");
    }
    printf("Transpose of the matrix\n");
    for(i=0;i<n;i++)
    {
        for(j=0;j<m;j++)
        {
            printf("%d ",b[i][j]);
        }
        printf("\n");
    }
    return 0;
}
