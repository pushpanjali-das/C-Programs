/* First letter of each  word in a string
    Input:
    Hey! Good Morning
    
    Output:
    H G M
*/

#include<stdio.h>
int main()
{
    char x[40];
    int i;
    printf("Enter the string\n");
    scanf("%[^\n]",x);
    printf("%c  ",x[0]);
    for(i=0;x[i]!='\0';i++)
    {
        if(x[i]==32)
        {
            printf("%c  ",x[i+1]);
        }
    }
    return 0;
}
