/* Program to Reverse all the numbers between 2 ranges */
#include<stdio.h>
int main()
{
    int n1,n2,temp,r,newnum;
    printf("enter the range\n");
    scanf("%d%d",&n1,&n2);
    while(n1<=n2)
    {
        temp=n1;
        newnum=0;
        while(temp>0)
        {
            r=temp%10;
            newnum=(newnum*10)+r;
            temp=temp/10;
        }
        printf("%d\t",newnum);
        n1++;
    }
    return 0;
}
