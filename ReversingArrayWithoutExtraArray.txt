/* Program to interchange (or) REVERSE (or) swap the elements of the array 

    WITHOUT USING EXTRA ARRAY
*/
#include<stdio.h>
int main()
{
    int arr[10],i,j,n,temp;
    printf("Enter the number of variables to be used\n");
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        printf("Enter value of arr[%d]\n",i);
        scanf("%d",&arr[i]);
    }
    printf("Given array is\n");
    for(i=0;i<n;i++)
    {
        printf("arr[%d]=%d\n",i,arr[i]);
    }
    for(i=0,j=n-1;i<j;i++,j--)
    {
        temp=arr[i];
        arr[i]=arr[j];
        arr[j]=temp;
    }
    printf("Reverse array is\n");
    for(i=0;i<n;i++)
    {
        printf("arr[%d]=%d\n",i,arr[i]);
    }
    return 0;
}
