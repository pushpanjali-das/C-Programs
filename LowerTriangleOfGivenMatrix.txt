/* Program to print Lower triangle. (Sqare matix only)
    
    1   2   3
    4   5   6
    7   8   9
    
Lower Triangle:
    +++++++++
    4               4(1,0)
    7   8           7(2,0)      8(2,1)
    
*/

#include<stdio.h>
int main()
{
    int arr[5][5],i,j,n;
    printf("Enter number of rows and columns\n");
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        for(j=0;j<n;j++)
        {
            printf("Enter value of arr[%d][%d]\n",i,j);
            scanf("%d",&arr[i][j]);
        }
    }
    printf("Given matrix\n");
    for(i=0;i<n;i++)
    {
        for(j=0;j<n;j++)
        {
            printf("%d ",arr[i][j]);
        }
        printf("\n");
    }
    printf("Lower triangle\n");
    for(i=0;i<n;i++)
    {
        for(j=0;j<n;j++)
        {
            if(i>j)
            {
                printf("%d ",arr[i][j]);
            }
            else
            {
                printf("  ");
            }
        }
        printf("\n");
    }
    return 0;
}
