//Finding Armstrong numbers using User Defined Function

include<stdio.h>
int digitsCount(int i)
{
    int c=0;
    while(i>0)
    {
        c++;
        i=i/10;
    }
    return c;
}
int power(int r,int count1)
{
    int Pow=1;
    while(count1>0)
    {
        Pow=Pow*r;
        count1--;
    }
    return Pow;
}
int main()
{
    int n1,n2,count1,sum=0,temp,r,p;
    printf("enter 2 intervals\n");
    scanf("%d%d",&n1,&n2);
    for(int i=n1;i<=n2;i++)
    {
        temp=i;
        sum=0;
        count1=digitsCount(temp);
        while(temp>0)
        {
            r=temp%10;
            p=power(r,count1);
            sum=sum+p;    
            temp=temp/10;
        }
        if(sum==i)
        {
            printf("%d\n",i);
        }
    }
    printf("program has ended successfully\n");
    return 0;
}
