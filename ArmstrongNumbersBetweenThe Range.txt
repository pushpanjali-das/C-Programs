/* Print all armstrong numbers between the range */
#include<stdio.h>
#include<math.h>
int main()
{
    int n1,n2,temp,r,sum,count,temp2;
    printf("enter the range\n");
    scanf("%d%d",&n1,&n2);
    while(n1<=n2)
    {
        temp=n1;
        temp2=n1;
        count=0;
        sum=0;
        while(temp>0)
        {
            count++;
            temp=temp/10;
        }
        while(temp2>0)
        {
            r=temp2%10;
            sum=sum+pow(r,count);
            temp2=temp2/10;
        }
        if(sum==n1)
        {
            printf("%d\t",n1);
        }
        n1++;
    }
    return 0;
}
