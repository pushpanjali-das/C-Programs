/* Program to concatenate (or) join (or) combine 2 strings */

#include<stdio.h>
#include<string.h>

int main()
{
    char x[20],y[20];
    int n;
    printf("Enter both the strings\n");
    scanf("%[^\n] %[^\n]",&x,&y);
    strcat(x,y);
    printf("New string is %s\n",x);
    printf("Enter the number of charcaters of 2nd string to be combined\n");
    scanf("%d",&n);
    strncat(x,y,n);
    printf("New string is %s",x);
    return 0;
}
