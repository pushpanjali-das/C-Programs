/*Program to interchange 2 numbers using pointers*/
#include<stdio.h>
int main()
{
    int a,b,*p,*q,temp;
    p=&a;
    q=&b;
    printf("Enter values of a and b\n");
    scanf("%d%d",p,q);
    printf("Before interchanging *p=%d and *q=%d\n",*p,*q);
    temp=*p;
    *p=*q;
    *q=temp;
    printf("After interchange *p=%d and *q=%d",*p,*q);
    return 0;
}
