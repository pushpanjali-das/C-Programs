/* Program to convert upper case characters in the string into lower case */

#include<stdio.h>
int main()
{
    char x[40];
    int i;
    printf("Enter the string\n");
    scanf("%[^\n]",&x);
    for(i=0;x[i]!='\0';i++)
    {
        if(x[i]>=65 && x[i]<=90)
        {
            x[i]=x[i]+32;
        }
    }
    printf("String lower case is %s",x);
    return 0;
}
