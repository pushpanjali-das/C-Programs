/* Program to check given numbers inside the given range are prime or not 

In this specific program, using pointers does not provide any significant advantage.
If you want a function to modify the value of variables in the caller's scope, pointers are essential.

*/
#include<stdio.h>
int main()
{
    int n1,n2,*p1,*p2,i,j=1,count=0;
    p1=&n1;
    p2=&n2;
    printf("Enter the range\n");
    scanf("%d%d",p1,p2);
    for(i=(*p1);i<=(*p2);i++)
    {
        count=0;
        for(j=1;j<=i;j++)
        {
            if(i%j==0)
            {
                count++;
            }
        }
        if(count==2)
        {
            printf("%d\n",i);
        }
    }
    return 0;
}
