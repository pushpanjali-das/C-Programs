/* Program to check whether the given n*n (or) m*n matrix is Symmetric or not.
    Condition:
             Matrix =  Matrix Transpose.
             
                                a(2*2) = 1 2
                                         2 1
            
    Transpose matrix           b(2*2) =  1 2
                                         2 1
*/

#include<stdio.h>
int main()
{
    int a[5][5],b[5][5],i,j,m,n,flag=1; //flag=1, assuming matrix is                                            Symmetric
    
    printf("Enter number of rows and columns\n");
    scanf("%d%d",&m,&n);
    for(i=0;i<m;i++)
    {
        for(j=0;j<n;j++)
        {
            printf("Enter value of a[%d][%d]\n",i,j);
            scanf("%d",&a[i][j]);
            b[j][i]=a[i][j];
        }
    }
    printf("Given matrix\n");
    for(i=0;i<m;i++)
    {
        for(j=0;j<n;j++)
        {
            printf("%d ",a[i][j]);
        }
        printf("\n");
    }
    printf("Tranpose matrix\n");
    for(i=0;i<n;i++)
    {
        for(j=0;j<m;j++)
        {
            printf("%d ",b[i][j]);
        }
        printf("\n");
    }
    for(i=0;i<n;i++)
    {
        for(j=0;j<n;j++)
        {
            if(a[i][j]!=b[i][j])
            {
                flag=0;
                break;
            }
        }
    }
    if(flag==1)
    {
        printf("Symmetric matrix");
    }
    else
    {
        printf("Not symmetric matrix");
    }
    return 0;
}
