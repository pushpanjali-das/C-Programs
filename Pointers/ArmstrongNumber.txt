/* Program to print armstrong numbers between a range*/
#include<stdio.h>
int main()
{
    int n1,n2,*p1,*p2,pow,r,sum,digits=0,count,dup1,dup2,i;
    p1=&n1;
    p2=&n2;
    printf("Enter the range\n");
    scanf("%d%d",p1,p2);
    for(i=(*p1);i<=(*p2);i++)
    {
        dup1=i;
        dup2=i;
        digits=0;
        while(dup1>0)
        {
            digits++;
            dup1=dup1/10;
        }
      //  printf("NUmber of digits %d\n",digits);
        sum=0;
        while(dup2>0)
        {
            count=digits;
            pow=1;
            r=dup2%10;
            while(count>0)
            {
                pow=pow*r;
                count--;
            }
            sum=sum+pow;
            dup2=dup2/10;
        }
        if(sum==i)
        {
            printf("%d\n",i);
        }
    }
    return 0;
}
