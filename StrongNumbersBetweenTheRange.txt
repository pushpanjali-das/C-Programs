/* Strong Number
    A number where the sum of factorials of individual digits present in the given numbers is eqauls to the original number
    EX: 145 = 1!+4!+5!
            = 1+24+120
            = 145 
*/

#include<stdio.h>
int main()
{
    int n1,n2,fact,sum=0,r,temp;
    printf("Enter the range\n");
    scanf("%d%d",&n1,&n2);
    while(n1<=n2)
    {
        temp=n1;
        sum=0;
        while(temp>0)
        {
            fact=1;
            r=temp%10;
            while(r>0)
            {
                fact=fact*r;
                r--;
            }
            sum=sum+fact;
            temp=temp/10;
        }
        if(sum==n1)
        {
            printf("%d\n",n1);
        }
        n1++;
    }
    return 0;
}
