/* Program to find the number of vowels present in the given file */
#include<stdio.h>
#include<conio.h>
int main()
{
	FILE *fp;
	int vowels=0;
	char c;
	clrscr();
	fp=fopen("sophia.txt","r ");
	while((c=getc(fp))!=-1)
	{
		if(c=='a'||c=='e'||c=='i'||c=='o'||c=='u'||c=='A'||c=='E'||c=='I'||c=='O'||c=='U')
		{
			vowels++;
		}
	}
	fclose(fp);
	printf("Number of vowels in the file is %d",vowels);
	return 0;
}
