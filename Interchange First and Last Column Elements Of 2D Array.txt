/* C Program to Interchange Elements of First and Last Columns of the given array
Input:
9 7 5
2 3 4
5 2 6

Output: 
5 7 9
4 3 2
6 2 5

*/
#include<stdio.h>
int main()
{
    int a[5][5],i,j,r,c,temp;
    printf("Enter the number of rows and columns\n");
    scanf("%d%d",&r,&c);
    printf("Enter array elements\n");
    for(i=0;i<r;i++)
    {
        for(j=0;j<c;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
    printf("Array before interchanging the 1st and last columns\n");
    for(i=0;i<r;i++)
    {
        for(j=0;j<c;j++)
        {
            printf("%d  ",a[i][j]);
        }
        printf("\n");
    }
    printf("Array after interchanging the 1st and last columns\n");
    for(i=0;i<r;i++)
    {
        for(j=0;j<c;j++)
        {
            if(j==0)
            {
                temp=a[i][j];
                a[i][j]=a[i][c-1];
                a[i][c-1]=temp;
            }
            printf("%d  ",a[i][j]);
        }
        printf("\n");
    }
    return 0;
}
