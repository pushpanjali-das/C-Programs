/* Program to replace an element in the array with the new element */
#include<stdio.h>

int main()
{
    int arr[10],i,n,ne,se,found=0;
    printf("Enter the number of variables to be used\n");
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        printf("Enter the value of arr[%d]\n",i);
        scanf("%d",&arr[i]);
    }
    
    printf("Enter the element which needs to be replaced by the new element\n");
    scanf("%d",&se);
    
    printf("Array before replacing with new element\n");
    for(i=0;i<n;i++)
    {
        printf("arr[%d]=%d\n",i,arr[i]);
    }
    
    for(i=0;i<n;i++)
    {
        if(arr[i]==se)
        {
            found=1;
            break;
        }
    }
    if(found==1)
    {
        printf("Enter the new element\n");
        scanf("%d",&ne);
        arr[i]=ne;
        printf("Array after replacing with new element is\n");
        for(i=0;i<n;i++)
        {
            printf("arr[%d]=%d\n",i,arr[i]);
        }
    }
    else
    {
        printf("Element can not be replaced as the searching element is not found");
    }
    return 0;
}
