/* Program to print the boundary elements of the 2D array */
#include<stdio.h>
int main()
{
    int a[5][5],i,j,r,c;
    printf("Enter the number of rows and columns\n");
    scanf("%d%d",&r,&c);
    for(i=0;i<r;i++)
    {
        for(j=0;j<c;j++)
        {
            printf("Enter value of a[%d][%d]\n",i,j);
            scanf("%d",&a[i][j]);
        }
    }
    for(i=0;i<r;i++)
    {
        for(j=0;j<c;j++)
        {
            if(i==0||i==r-1||j==0||j==c-1)
            {
                printf("%-3d",a[i][j]);
            }
            else
            {
                printf("   ");
            }
        }
        printf("\n");
    }
    return 0;
}
