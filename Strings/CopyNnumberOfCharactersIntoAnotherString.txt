/* Program to copy "n" charcaters of a string into another string */
#include<stdio.h>
int main()
{
    char x[40],y[40];
    int i=0,n;
    printf("Enter the string\n");
    scanf("%[^\n]",&x);
    printf("Enter the number of characters to be copied\n");
    scanf("%d",&n);
    while(i<n && x[i]!='\0')
    {
        y[i]=x[i];
        i++;
    }
    printf("New string is %s",y);
    return 0;
}
