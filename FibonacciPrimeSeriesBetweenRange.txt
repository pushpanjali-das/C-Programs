/* Program to print fibonacci prime series within range */
#include<stdio.h>
int main()
{
    int a=0,b=1,c,n1,n2,i,count;
    printf("Enter lower and upper limit values\n");
    scanf("%d%d",&n1,&n2);
    if(n1==0)
    {
        printf("%d %d ",a,b);
    }
    else if(n1==1)
    {
        printf("%d ",b);
    }
    do
    {
        c=a+b;
        if((c>=n1)&&(c<=n2))
        {
            i=1,count=0;
            while(i<=c)
            {
                if(c%i==0)
                {
                    count++;
                }
                i++;
            }
            if(count==2)
            {
                printf("%d ",c);
            }
        }
        a=b;
        b=c;
    }while(c<=n2);
    return 0;
}
