/* Program to print fibonacci series upto given 'n' value */
#include<stdio.h>
int main()
{
    int a=0,b=1,c,n;
    printf("Enter the number\n");
    scanf("%d",&n);
    printf("%d %d ",a,b);
    do
    {
        c=a+b;
        if(c<=n)
        {
            printf("%d ",c);
        }
        a=b;
        b=c;
    }while(c<=n);
    return 0;
}
