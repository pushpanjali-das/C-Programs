/* Count of all, even and odd factors of a given number */
#include<stdio.h>
int main()
{
    int n,i=1,count=0,counto=0,counte=0;
    printf("enter the number\n");
    scanf("%d",&n);
    while(i<=n)
    {
        if(n%i==0)
        {
            printf("%d\n",i);
            count++;
            if(i%2==1)
            {
                counto++;
            }
            else
            {
                counte++;
            }
        }
        i++;
    }
    printf("For the number %d count of all the factors is %d\nCount of odd factors is %d\nCount of even factor is %d",n,count,counto,counte);
}
