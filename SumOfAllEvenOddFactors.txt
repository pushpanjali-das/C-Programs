/* Sum of all, even and odd factors of a given number */
#include<stdio.h>
int main()
{
    int n,i=1,sum=0,sume=0,sumo=0;
    printf("Enter the number\n");
    scanf("%d",&n);
    while(i<=n)
    {
        if(n%i==0)
        {
            printf("%d\n",i);
            sum=sum+i;
            if(i%2==0)
            {
                sume=sume+i;
            }
            else
            {
                sumo=sumo+i;
            }
        }
        i++;
    }
    printf("For the number %d sum of all factors is %d\nSum of all even factors is %d\nSum of all odd fcactors is %d",n,sum,sume,sumo);
    return 0;
}
