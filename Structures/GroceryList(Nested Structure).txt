#include<stdio.h>
struct items
{
    char itemsname[20];
};
struct category
{
    char name[20];
    int itemscount;
    struct items itemslist[10];
};

int main()
{
    struct category c[10];
    int nc,i,j;
    printf("Enter the number of categories available\n");
    scanf("%d",&nc);
    for(i=0;i<nc;i++)
    {
        printf("Enter the name of category %d\n",i+1);
        scanf("%s",c[i].name);
        printf("Enter number of items in %s\n",c[i].name);
        scanf("%d",&c[i].itemscount);
        for(j=0;j<c[i].itemscount;j++)
        {
            printf("Enter the item %d\n",j+1);
            scanf("%s",c[i].itemslist[j].itemsname);
        }
    }
    printf("******Grocery list*******\n");
    for(i=0;i<nc;i++)
    {
        printf("%d.%s\n",i+1,c[i].name);
        for(j=0;j<c[i].itemscount;j++)
        {
            printf("%d.%s\n",j+1,c[i].itemslist[j].itemsname);
        }
        printf("\n");
    }
    return 0;
}
