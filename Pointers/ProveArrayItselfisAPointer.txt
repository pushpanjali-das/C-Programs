/* Progarm to prove array itself is a pointer */
#include<stdio.h>
int main()
{
    int a[5]={11,22,33,44,55},i;
    for(i=0;i<5;i++)
    {
        printf("Address of a[%d] is %u and value is %d\n",i,a+i,*(a+i));
    }
    //Here to move to the next address, everytime "i" is getting added to the base address. 
    Base address does not modify at any stage while traversing it still ramains the same.
    i.e., We are not assiging a+i value to a again. (a!=a+i)
    printf("base address is %u",a);
    return 0;
}
