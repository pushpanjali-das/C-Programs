#include<stdio.h>
void removed(int arr[],int num,int value)
{
    for(int i=0;i<num;i++)
    {
        if(arr[i]!=value)
        {
            printf("%d ",arr[i]);
        }
    }
}
int main()
{
    int arr[5];
    int e;
    printf("enter elements of array\n");
    for(int i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    printf("enter element to be searched and to be removedfrom the array\n");
    scanf("%d",&e);

}
