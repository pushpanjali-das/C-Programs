/* Program to find the length of the string */

#include<stdio.h>
int main()
{
    char x[40];
    int i=0;
    printf("Enter the string\n");
    scanf("%[^\n]",&x);                 //a n j a l i \0              
                                        //0 1 2 3 4 5  6
    while(x[i]!='\0')                   //x[6]!='\0' --> false, i=6
    {
        i++;
    }
    printf("Length of the string is %d",i);
    return 0;
}
