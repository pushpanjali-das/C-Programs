/* Program to check the given string is palindrome or not */
#include<stdio.h>
int main()
{
    char x[40],y[40];
    int i,j,l=0;
    printf("Enter the string\n");
    scanf("%[^\n]",&x);
    while(x[l]!='\0')
    {
        l++;
    }
    for(i=0,j=l-1;i<=l-1,j>=0;i++,j--)
    {
        y[i]=x[j];
    }
    i=0;
    while(i<l)
    {
        if(x[i]!=y[i])
        {
            break;
        }
        else
        {
            i++;
        }
    }
    if(i==l)
    {
        printf("%s is palindrome",x);
    }
    else
    {
        printf("%s is not palindrome",x);
    }
    return 0;
}
