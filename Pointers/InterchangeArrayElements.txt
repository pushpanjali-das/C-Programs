/* Program to interchange the values of array */
#include<stdio.h>
int main()
{
    int a[10],n,*p,*q,temp;
    printf("Enter the number of variables to be used\n");
    scanf("%d",&n);
    printf("Enter %d values\n",n);
    for(p=&a[0];p<&a[n];p++)
    {
        scanf("%d",p);
    }
    for(p=a,q=&a[n-1];p<q;p++,q--)
    {
        temp=*p;
        *p=*q;
        *q=temp;
    }
    printf("Array after interchanging the elements is\n");
    for(p=a;p<&a[n];p++)
    {
        printf("%d\t",*p);
    }
    return 0;
}
